  hud.querySelector('#world-remove').setAttribute('aria-pressed', String(worldState.buildMode === 'delete'));
}

function sendWorld(payload) {
  const socket = network.socket;
  if (!network.ready || !socket || socket.readyState !== WebSocket.OPEN) return false;
  socket.send(JSON.stringify(payload));
  return true;
}

function claimCurrentChunk() {
  const current = currentWorldChunk();
  if (!sendWorld({ type: 'world:claim', chunkX: current.chunkX, chunkZ: current.chunkZ })) {
    worldFeedback('World server is not connected.', 'error');
  } else {
    worldFeedback('Claim request sent…');
  }
}

function actionId() {
  return crypto.randomUUID ? crypto.randomUUID() : 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, c => {
    const r = Math.random() * 16 | 0;
    return (c === 'x' ? r : (r & 3 | 8)).toString(16);
  });
}

function normalizeChunk(record) {
  const chunkX = Number(record?.chunkX ?? record?.chunk_x);
  const chunkZ = Number(record?.chunkZ ?? record?.chunk_z);
  return {
    worldId: record?.worldId ?? record?.world_id ?? worldState.id,
    chunkX, chunkZ,
    ownerId: record?.ownerId ?? record?.owner_id ?? null,
    claimedAt: record?.claimedAt ?? record?.claimed_at ?? null,
    voxelData: record?.voxelData ?? record?.voxel_data ?? {},
    revision: Number(record?.revision) || 0,
    updatedAt: record?.updatedAt ?? record?.updated_at ?? null
  };
}

function applyWorldChunk(record) {
  const chunk = normalizeChunk(record);
  if (!Number.isFinite(chunk.chunkX) || !Number.isFinite(chunk.chunkZ)) return;
  const key = worldChunkKey(chunk.chunkX, chunk.chunkZ);
  worldState.chunks.set(key, chunk);
  if (worldOwnsChunk(chunk)) worldState.claims.set(key, chunk);
  else worldState.claims.delete(key);
  if (terrain?.chunks?.has(key)) rebuildWorldChunk(chunk.chunkX, chunk.chunkZ);
  worldState.chunkBorderSignature = '';
  updateWorldHud();
}

function rebuildWorldChunk(cx, cz) {
  const key = worldChunkKey(cx, cz);
  const previous = terrain.chunks.get(key);
  if (previous) {
    gl.deleteBuffer(previous.instanceBuffer);
    gl.deleteVertexArray(previous.vao);
  }
  terrain.chunks.set(key, createTerrainChunk(cx, cz));
}

function worldVoxelEntry(worldX, y, worldZ) {
  const chunkX = Math.floor(worldX / CHUNK_SIZE);
  const chunkZ = Math.floor(worldZ / CHUNK_SIZE);
  const localX = positiveLocal(worldX);
  const localZ = positiveLocal(worldZ);
  const record = worldState.chunks.get(worldChunkKey(chunkX, chunkZ));
  const key = String(localX) + ':' + String(y) + ':' + String(localZ);
  return { record, value: record?.voxelData?.[key], chunkX, chunkZ, localX, localZ, key };
}

function baseTerrainHeightForBuild(worldCellX, worldCellZ) {
  let height = getBaseTerrainHeight(worldCellX, worldCellZ);
  const distance = Math.hypot(worldCellX, worldCellZ);
  if (distance < 7) {
    const centerHeight = getBaseTerrainHeight(0, 0);
    const blend = smoothstep(2.0, 7.0, distance);
    height = Math.round(mix(centerHeight, height, blend));
  }
  return clamp(height, MIN_HEIGHT, MAX_HEIGHT);
}

function baseVoxelType(worldX, y, worldZ, height = baseTerrainHeightForBuild(worldX, worldZ)) {
  if (y === height - 1) return getSurfaceType(worldX, worldZ, height);
  return height - 1 - y <= 2 ? BLOCK_DIRT : BLOCK_STONE;
}

function occupiedVoxelAt(worldX, y, worldZ) {
  if (![worldX, y, worldZ].every(Number.isFinite)) return null;
  const entry = worldVoxelEntry(worldX, y, worldZ);
  if (entry.value?.action === 'remove') return null;
  if (entry.value?.action === 'place') {
    return {
      kind: 'placed',
      type: BLOCK_NAME_TO_TYPE[entry.value?.block?.type] ?? BLOCK_STONE,
      ...entry
    };
  }
  const height = baseTerrainHeightForBuild(worldX, worldZ);
  if (y >= WORLD_FLOOR && y < height) {
    return { kind: y === WORLD_FLOOR ? 'bedrock' : 'terrain', type: baseVoxelType(worldX, y, worldZ, height), ...entry };
  }
  return null;
}

function reconcileWorldVoxels(instances, cx, cz) {
  const cells = new Map();
  for (let index = 0; index < instances.length; index += 4) {
    const worldX = Math.floor(instances[index]);
    const y = Math.floor(instances[index + 1]);
    const worldZ = Math.floor(instances[index + 2]);
    cells.set(worldX + ':' + y + ':' + worldZ, [instances[index], instances[index + 1], instances[index + 2], instances[index + 3]]);
  }

  const startX = cx * CHUNK_SIZE;
  const startZ = cz * CHUNK_SIZE;
  for (let localZ = 0; localZ < CHUNK_SIZE; localZ += 1) {
    for (let localX = 0; localX < CHUNK_SIZE; localX += 1) {
      const worldX = startX + localX;
      const worldZ = startZ + localZ;
      const height = baseTerrainHeightForBuild(worldX, worldZ);
      for (let y = WORLD_FLOOR; y < height; y += 1) {
        const key = worldX + ':' + y + ':' + worldZ;
        const occupied = occupiedVoxelAt(worldX, y, worldZ);
        if (!occupied) cells.delete(key);
        else cells.set(key, [worldX + 0.5, y + 0.5, worldZ + 0.5, occupied.type]);
      }
    }
  }

  const record = worldState.chunks.get(worldChunkKey(cx, cz));
  for (const [key, value] of Object.entries(record?.voxelData || {})) {
    const [localX, y, localZ] = key.split(':').map(Number);
    if (![localX, y, localZ].every(Number.isFinite)) continue;
    const worldX = startX + localX;
    const worldZ = startZ + localZ;
    const worldKey = worldX + ':' + y + ':' + worldZ;
    if (value?.action === 'remove') cells.delete(worldKey);
    if (value?.action === 'place') {
      cells.set(worldKey, [worldX + 0.5, y + 0.5, worldZ + 0.5, BLOCK_NAME_TO_TYPE[value?.block?.type] ?? BLOCK_STONE]);
    }
  }

  instances.length = 0;
  for (const instance of cells.values()) instances.push(...instance);
}

function overlayColumnHeight(worldCellX, worldCellZ, baseHeight) {
  let highestCandidate = Math.max(baseHeight, WORLD_FLOOR + 1);
  const entry = worldVoxelEntry(worldCellX, 0, worldCellZ);
  for (const [key, value] of Object.entries(entry.record?.voxelData || {})) {
    const [localX, y, localZ] = key.split(':').map(Number);
    if (localX === entry.localX && localZ === entry.localZ && value?.action === 'place') {
      highestCandidate = Math.max(highestCandidate, y + 1);
    }
  }
  for (let y = Math.ceil(highestCandidate) - 1; y >= WORLD_FLOOR; y -= 1) {
    if (occupiedVoxelAt(worldCellX, y, worldCellZ)) return y + 1;
  }
  return WORLD_FLOOR + 1;
}

function supportHeightAt(x, z, referenceY) {
  const worldX = Math.floor(x);
  const worldZ = Math.floor(z);
  const maximumTop = Math.floor(Number(referenceY) + MAX_STEP_HEIGHT + 0.001);
  for (let y = maximumTop - 1; y >= WORLD_FLOOR; y -= 1) {
    if (occupiedVoxelAt(worldX, y, worldZ)) return y + 1;
  }
  return WORLD_FLOOR + 1;
}

function canOccupyPlayerAt(x, z, groundY) {
  const minY = Math.floor(groundY + 0.04);
  const maxY = Math.floor(groundY + PLAYER_HEIGHT - 0.04);
  for (const [offsetX, offsetZ] of PLAYER_COLLISION_SAMPLES) {
    const worldX = Math.floor(x + offsetX);
    const worldZ = Math.floor(z + offsetZ);
    for (let y = minY; y <= maxY; y += 1) {
      if (occupiedVoxelAt(worldX, y, worldZ)) return false;
    }
  }
  return true;
}

function resolvePlayerGround(x, z, referenceY = player.groundY || player.y) {
  return supportHeightAt(x, z, Number.isFinite(referenceY) ? referenceY : getTerrainHeight(x, z));
}

function voxelIntersectsPlayer(worldX, y, worldZ) {
  const voxelMinX = worldX;
  const voxelMaxX = worldX + 1;
  const voxelMinZ = worldZ;
  const voxelMaxZ = worldZ + 1;
  const playerMinX = player.x - PLAYER_RADIUS;
  const playerMaxX = player.x + PLAYER_RADIUS;
  const playerMinZ = player.z - PLAYER_RADIUS;
  const playerMaxZ = player.z + PLAYER_RADIUS;
  const playerMinY = player.y + 0.02;
  const playerMaxY = player.y + PLAYER_HEIGHT - 0.02;
  return voxelMaxX > playerMinX && voxelMinX < playerMaxX
    && voxelMaxZ > playerMinZ && voxelMinZ < playerMaxZ
    && y + 1 > playerMinY && y < playerMaxY;
}

function invertWorldMatrix(out, a) {
  const a00=a[0],a01=a[1],a02=a[2],a03=a[3],a10=a[4],a11=a[5],a12=a[6],a13=a[7];
  const a20=a[8],a21=a[9],a22=a[10],a23=a[11],a30=a[12],a31=a[13],a32=a[14],a33=a[15];
  const b00=a00*a11-a01*a10,b01=a00*a12-a02*a10,b02=a00*a13-a03*a10,b03=a01*a12-a02*a11;
  const b04=a01*a13-a03*a11,b05=a02*a13-a03*a12,b06=a20*a31-a21*a30,b07=a20*a32-a22*a30;
  const b08=a20*a33-a23*a30,b09=a21*a32-a22*a31,b10=a21*a33-a23*a31,b11=a22*a33-a23*a32;
  let det=b00*b11-b01*b10+b02*b09+b03*b08-b04*b07+b05*b06;
  if (!det) return false;
  det=1/det;
  out[0]=(a11*b11-a12*b10+a13*b09)*det; out[1]=(a02*b10-a01*b11-a03*b09)*det;
  out[2]=(a31*b05-a32*b04+a33*b03)*det; out[3]=(a22*b04-a21*b05-a23*b03)*det;
  out[4]=(a12*b08-a10*b11-a13*b07)*det; out[5]=(a00*b11-a02*b08+a03*b07)*det;
  out[6]=(a32*b02-a30*b05-a33*b01)*det; out[7]=(a20*b05-a22*b02+a23*b01)*det;
  out[8]=(a10*b10-a11*b08+a13*b06)*det; out[9]=(a01*b08-a00*b10-a03*b06)*det;
